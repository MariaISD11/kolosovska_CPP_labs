#include <iostream>
#include <cstdlib>
#include <ctime>
using namespace std;

int main()
{
// 3

    srand(time(0));
    const int n = 10;
    int arr[n];

    cout << "Початковий масив: ";
    for (int i = 0; i < n; ++i) {
        arr[i] = rand() % 100;
        cout << arr[i] << " ";
    }
    cout << endl;

    int min_idx = 0, max_idx = 0;
    for (int i = 1; i < n; ++i) {
        if (arr[i] < arr[min_idx]) {
            min_idx = i;
        }
        if (arr[i] > arr[max_idx]) {
            max_idx = i;
        }
    }

    swap(arr[min_idx], arr[max_idx]);

    cout << "Новий масив: ";
    for (int i = 0; i < n; ++i) {
        cout << arr[i] << " ";
    }
    cout << endl;

// 4

    srand(time(0));
    const int n1 = 10;
    int arr1[n1];

    cout << "Масив: ";
    for (int i = 0; i < n1; ++i) {
        arr1[i] = rand() % 100; 
        cout << arr1[i] << " ";
    }
    cout << endl;

    int A;
    cout << "Введіть число A: ";
    cin >> A;

    int count = 0;
    for (int i = 0; i < n1; ++i) {
        if (arr[i] > A) {
            count++;
        }
    }

    cout << "Кількість елементів, більших за " << A << ": " << count << endl;

// 5

    srand(time(0));
    const int n2 = 10;
    int arr2[n2];

    for (int i = 0; i < n2; i++) {
        arr2[i] = rand() % 100 + 1;

        for (int j = 0; j < i; j++) {
            if (arr2[i] == arr[j]) {
                i--;
                break;
            }
        }
    }

    cout << "Масив: ";
    for (int i = 0; i < n2; i++) {
        cout << arr2[i] << " ";
    }

    int max1 = arr2[0];
    int max2 = arr2[1];

    if (max2 > max1) {
        swap(max1, max2);
    }

    for (int i = 2; i < n2; i++) {
        if (arr2[i] > max1) {
            max2 = max1;
            max1 = arr2[i];
        }
        else if (arr2[i] > max2) {
            max2 = arr2[i];
        }
    }

    cout << "\nДругий за величиною елемент: " << max2;

// 6

    }

    int zero_index = -1;
    for (int i = 0; i < n3; ++i) {
        if (arr3 == 0) {
            zero_index = i;
            break; 
        }
    }

    if (zero_index != -1) {
        cout << "Індекс першого нульового елемента: " << zero_index << endl;
    } else {
        cout << "Масив не містить нульових елементів." << endl;
    }
    
    return 0;
}
