    srand(time(0));
    const int n = 10;
    int arr[n];

    cout << "Початковий масив: ";
    for (int i = 0; i < n; ++i) {
        arr[i] = rand() % 100;
        cout << arr[i] << " ";
    }
    cout << endl;

    int min_idx = 0, max_idx = 0;
    for (int i = 1; i < n; ++i) {
        if (arr[i] < arr[min_idx]) {
            min_idx = i;
        }
        if (arr[i] > arr[max_idx]) {
            max_idx = i;
        }
    }

    swap(arr[min_idx], arr[max_idx]);

    cout << "Новий масив: ";
    for (int i = 0; i < n; ++i) {
        cout << arr[i] << " ";
    }
    cout << endl;

                i--;

            }
        }

    if (max2 > max1) {

        if (arr[i] > max1) {
            max2 = max1;
            max1 = arr[i];
        }

        cout << "Індекс першого нульового елемента: " << zero_index << endl;
