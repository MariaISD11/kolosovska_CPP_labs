
#include <iostream>
#include <algorithm>
using namespace std;

int main()
{
// 1

    const int SIZE = 12;
    int m[SIZE] = {45, 12, 89, 3, 23, 78, 56, 9, 34, 67, 1, 90};

    // Виводимо початковий масив
    cout << "Початковий масив: " << endl;
    for (int i = 0; i < SIZE; i++) {
        cout << m[i] << " ";
    }
    cout << endl << endl;

    // Упорядкування за зростанням
    sort(m, m + SIZE);

    cout << "Масив за зростанням: " << endl;
    for (int i = 0; i < SIZE; i++) {
        cout << m[i] << " ";
    }
    cout << endl << endl;

    // Упорядкування за спаданням
    sort(m, m + SIZE, greater<int>());

    cout << "Масив за спаданням: " << endl;
    for (int i = 0; i < SIZE; i++) {
        cout << m[i] << " ";
    }
    cout << endl;

// 2

    const int SIZE = 10;
    int arr[SIZE] = {-5, 12, -3, 8, 15, -2, 15, -7, 4, -1};

    cout << "Початковий масив:" << endl;
    for (int i = 0; i < SIZE; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;

    int maxIndex = 0;
    int maxValue = arr[0];

    for (int i = 1; i < SIZE; i++) {
        if (arr[i] > maxValue) {
            maxValue = arr[i];
            maxIndex = i;
        }
    }

    cout << "Максимальний елемент: " << maxValue << " (його індекс: " << maxIndex << ")" << endl;

    for (int i = 0; i < maxIndex; i++) {
        if (arr[i] < 0) {
            arr[i] = 0;
        }
    }

    cout << "Масив після заміни:" << endl;
    for (int i = 0; i < SIZE; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;

    return 0;
}
