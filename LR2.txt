#include <iostream>
#include <cmath>
using namespace std;

int main(){
	double x, z;

	cout << "Type an x: ";
	cin >> x;

	if (x >= 0){
		z = cos(x) - pow(sin(x), 3);}
	else {
		z = x * exp(-x) + 1.0 / (x-1.3);}

	cout << "Z = " << z;
	return 0;
}
